Prompt 5 · a fórmula mora num lugar só

12 KPIs governados no Unity Catalog

-- metric view: definida uma vez
measures:
  - name: spi
    expr: SUM(ev) / SUM(pv)
  - name: cpi
    expr: SUM(ev) / SUM(ac)
  - name: estouro_previsto
    expr: SUM(orcamento / cpi - orcamento)

-- usada em qualquer lugar
SELECT uf, MEASURE(cpi)
FROM gold.mv_obras GROUP BY uf
